## Phase 2 — Fix the Vite Config for Production Builds

Figma Make's environment does not use standard Vite build output. Before deploying, verify `vite.config.ts` has a standard configuration:

```bash
cat vite.config.ts
```

It should look like this — if it has Figma-specific overrides, replace them:

```ts
// vite.config.ts
import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";
import tailwindcss from "@tailwindcss/vite";

export default defineConfig({
  plugins: [react(), tailwindcss()],
  build: {
    outDir: "dist",
    sourcemap: true,
  },
});
```

Update `package.json` scripts:

```json
{
  "scripts": {
    "dev":              "vite",
    "build":            "pnpm build:tokens && vite build",
    "preview":          "vite preview",
    "build:tokens":     "node --import tsx/esm style-dictionary.config.ts",
    "build:tokens:watch": "node --watch --import tsx/esm style-dictionary.config.ts"
  }
}
```

Update the CSS entry from Figma's virtual entrypoint to the standard Vite entry:

```bash
# src/main.tsx should exist with:
cat src/main.tsx
```

If it doesn't exist, create it:

```tsx
// src/main.tsx
import { StrictMode } from "react";
import { createRoot } from "react-dom/client";
import "./styles/globals.css";
import App from "./app/App";

createRoot(document.getElementById("root")!).render(
  <StrictMode>
    <App />
  </StrictMode>
);
```

Create `index.html` at the project root if it doesn't exist:

```html
<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>MoneyPlugHub OS</title>
  </head>
  <body>
    <div id="root"></div>
    <script type="module" src="/src/main.tsx"></script>
  </body>
</html>
```

Test the build locally:

```bash
pnpm build
pnpm preview
# visit http://localhost:4173
```

### 3.4 Create the database schema

Go to Supabase Dashboard → SQL Editor → New query, then run:

```sql
-- ── Users & auth ─────────────────────────────────────────────────────────────
-- Supabase provides auth.users automatically. We extend it:

create table public.profiles (
  id          uuid primary key references auth.users(id) on delete cascade,
  handle      text unique not null,
  display_name text,
  avatar_url  text,
  tier        text default 'Bronze' check (tier in ('Bronze','Silver','Gold','Platinum')),
  level       int default 1,
  created_at  timestamptz default now()
);

alter table public.profiles enable row level security;

create policy "Users can read all profiles"
  on public.profiles for select using (true);

create policy "Users can update own profile"
  on public.profiles for update using (auth.uid() = id);

-- ── XP events ────────────────────────────────────────────────────────────────
create table public.xp_events (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid references public.profiles(id) on delete cascade,
  label       text not null,
  xp          int not null check (xp > 0),
  icon_type   text default 'share' check (icon_type in ('share','referral','quest')),
  created_at  timestamptz default now()
);

alter table public.xp_events enable row level security;

create policy "Users can read own XP events"
  on public.xp_events for select using (auth.uid() = user_id);

create policy "Service role can insert XP events"
  on public.xp_events for insert with check (true);

-- ── Computed XP summary view ──────────────────────────────────────────────────
create view public.xp_summary as
select
  p.id                                        as user_id,
  coalesce(sum(e.xp), 0)                      as current_xp,
  p.level,
  (p.level * 1000 + 500)                      as level_xp_goal,
  coalesce(sum(e.xp) filter (
    where e.created_at > now() - interval '1 day'
  ), 0)                                       as today_xp
from public.profiles p
left join public.xp_events e on e.user_id = p.id
group by p.id, p.level;

-- ── Referrals ─────────────────────────────────────────────────────────────────
create table public.referrals (
  id          uuid primary key default gen_random_uuid(),
  referrer_id uuid references public.profiles(id) on delete cascade,
  name        text not null,
  platform    text not null,
  status      text default 'Pending' check (status in ('Approved','Pending','Rejected')),
  reward      text default '$0',
  created_at  timestamptz default now()
);

alter table public.referrals enable row level security;

create policy "Users can read own referrals"
  on public.referrals for select using (auth.uid() = referrer_id);

create policy "Users can insert referrals"
  on public.referrals for insert with check (auth.uid() = referrer_id);

-- ── Leaderboard view ──────────────────────────────────────────────────────────
create view public.leaderboard_global as
select
  rank() over (order by sum(e.xp) desc) as rank,
  p.handle                              as handle,
  p.display_name                        as name,
  p.avatar_url                          as avatar,
  coalesce(sum(e.xp), 0)               as xp,
  '$' || coalesce(sum(r.reward_cents) / 100, 0)::text as earned,
  0                                     as change
from public.profiles p
left join public.xp_events e on e.user_id = p.id
left join public.referrals r on r.referrer_id = p.id and r.status = 'Approved'
group by p.id, p.handle, p.display_name, p.avatar_url;

-- ── Earnings (daily aggregate) ────────────────────────────────────────────────
create table public.earnings_daily (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid references public.profiles(id) on delete cascade,
  day         text not null,           -- e.g. 'Mon'
  value       numeric(10,2) default 0,
  date        date not null default current_date
);

alter table public.earnings_daily enable row level security;

create policy "Users can read own earnings"
  on public.earnings_daily for select using (auth.uid() = user_id);
```

## Phase 7 — CI/CD via GitHub Actions

The workflow at `.github/workflows/docs-deploy.yml` handles docs. Add a separate deploy workflow:

```bash
mkdir -p .github/workflows
```

```yaml
# .github/workflows/deploy.yml
name: Build & Deploy

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:
    name: Build
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: "22"

      - uses: pnpm/action-setup@v3
        with:
          version: 10

      - run: pnpm install --frozen-lockfile

      - name: Build tokens
        run: pnpm build:tokens

      - name: Build app
        run: pnpm exec vite build
        env:
          VITE_SUPABASE_URL: ${{ secrets.VITE_SUPABASE_URL }}
          VITE_SUPABASE_ANON_KEY: ${{ secrets.VITE_SUPABASE_ANON_KEY }}

      - uses: actions/upload-artifact@v4
        with:
          name: dist
          path: dist/

  deploy:
    name: Deploy to Vercel
    needs: build
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
    steps:
      - uses: actions/checkout@v4

      - uses: actions/download-artifact@v4
        with:
          name: dist
          path: dist/

      - name: Deploy
        run: npx vercel deploy --prebuilt --prod --token=${{ secrets.VERCEL_TOKEN }}
        env:
          VERCEL_ORG_ID: ${{ secrets.VERCEL_ORG_ID }}
          VERCEL_PROJECT_ID: ${{ secrets.VERCEL_PROJECT_ID }}
```

Add secrets to GitHub → repo → Settings → Secrets and variables → Actions:

```
VITE_SUPABASE_URL       → https://xxxxxxxxxxxx.supabase.co
VITE_SUPABASE_ANON_KEY  → eyJhbGci...
VERCEL_TOKEN            → from vercel.com/account/tokens
VERCEL_ORG_ID           → from .vercel/project.json after `vercel link`
VERCEL_PROJECT_ID       → from .vercel/project.json after `vercel link`
```

## Full Command Sequence (Copy-Paste)

```bash
# 0. Confirm node/pnpm
node --version && pnpm --version

# 1. Git init & push
git init && git add . && git commit -m "feat: initial prototype"
git remote add origin https://github.com/YOUR_ORG/moneyphubhub-os.git

# 2. Fix build
pnpm build          # must succeed with exit 0 and produce dist/

# 3. Create .env.local
echo "VITE_SUPABASE_URL=https://YOUR_PROJECT.supabase.co" >> .env.local
echo "VITE_SUPABASE_ANON_KEY=YOUR_ANON_KEY" >> .env.local

# 4. Run Supabase schema (paste SQL into Dashboard → SQL Editor)

# 5. Flip mock toggle
sed -i 's/USE_MOCK = true/USE_MOCK = false/' src/lib/api/mockData.ts

# 6. Test with real data
pnpm dev

# 7. Build for production
pnpm build

# 8. Deploy
vercel --prod

git add .github/
git commit -m "ci: add build & deploy workflow"
```
